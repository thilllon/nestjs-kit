## 2. Package layout and public API

One npm package, `nestjs-slightly-better-auth`. Every framework integration and every better-auth plugin integration lives in its own subpath. Optional peers are therefore imported statically, and only by the subpath that needs them. There is no `loadPackage`, no `createRequire` of peers and no top-level await (R:packaging §6; R:nest-di §8).

### 2.2 Exported symbols with signatures

Types derived from the user's instance are always computed from `typeof auth`, never re-declared (§11). Signatures are normative; bodies are illustrative.

  originCheck?: OriginCheckOptions;

export interface CookieForwardingOptions {

   * Forward Set-Cookie produced by direct auth.api.* calls that application code makes inside a handler, when the call
   * carries the caller's own credential or none. Default false, which is better-auth's server-call semantics
   * (a server call never touches the browser unless you ask). Per handler: @ForwardAuthCookies(). The library's own calls
   * (principal sources and policies) always forward. Handlers with forwarding on get better-auth's form-CSRF origin rule on unsafe
   * operations (§7.10), so turning this on instance-wide applies it everywhere. v3's per-call-site opt-in is gone: it forwarded
   * without the form check (§7.5). [R3:SEC-r3-04]
   */
  forwardDirectCalls?: boolean;
}

export interface OriginCheckOptions {
  /**
   * 'cookie' (default): an unsafe operation whose browser leg carries a cookie must come from a trusted origin, whatever credential
   * authenticated it, exactly as on better-auth's routes (§7.10). 'off': disabled (B19 warns in production). [R2:SEC-r2-01]

  mode?: "cookie" | "off";
  /**
   * A cookie-authenticated unsafe operation without Origin or Referer.

   * them; under 'reject' they need an Origin header. See §17 Q15. [R3:SEC-r3-03]
   */
  missingOrigin?: "reject" | "allow-non-browser";

 * Transport-neutral authentication + authorization (§7, §8). Singleton. It has no subclass seam: company-wide requirements are the
 * declarative `defaultRequirements` option, which every reader of a plan sees (§7.6). [R2:NEST-r2-08]
 */
export declare class BetterAuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): Promise<boolean>;
}
/** Compiled once per (controller class, handler), never per handler alone (§7.6). [R2:NEST-r2-01] */
export interface RoutePlan {
  readonly instance: string;
  /**
   * 'inherit': a handler a transport declares as reached only through an already authorized operation (GraphQL field resolvers
   * without method-level access, acceptance or requirement metadata; class-level metadata never applies to them, §7.6 step 3). The
   * guard does no I/O: it checks that an enclosing invocation recorded a reading and returns true, and readers take the nearest
   * enclosing invocation's reading (§7.8). [R2:NEST-r2-04] [R3:NEST-r3-05] [R4:NEST-r4-01]
   */
  readonly access: "public" | "optional" | "required" | "inherit";
  /** A transport's defaultAccessFor answered 'inherit' for the handler (a nested handler, §7.6 step 3), whatever its effective access. [R4:NEST-r4-03] */
  readonly nested: boolean;
  /**
   * defaultRequirements first (unless @SkipDefaultRequirements() applies), then class requirements (base first; not for handlers a
   * transport nests), then method requirements; empty unless access is 'required'. [R3:NEST-r3-05, NEST-r3-06]
   */
  readonly requirements: readonly RequirementExpr[];
  /** @SkipDefaultRequirements() removed the instance's defaultRequirements from this plan (counted in the boot summary, B22). [R3:NEST-r3-06] */
  readonly skipsDefaultRequirements: boolean;
  /**
   * The handler declares something in B16's sense (§5.4), computed with the same scoping as the plan: method-level metadata, class-level
   * metadata unless a transport nests the handler, and direct-call forwarding (@ForwardAuthCookies() or cookies.forwardDirectCalls).
   * The other instance-wide defaults (defaultAccess, the default kinds, defaultRequirements) never count. [R3:NEST-r3-05, NEST-r3-07]
   */
  readonly declares: boolean;
  /** 'authoritative' when the route asks for it, session.freshness says so, or a requirement demands fresh identity. */
  readonly freshness: "default" | "authoritative";
  /** Principal kinds admitted on the route (§7.6): @AcceptPrincipals or the instance's default kinds, plus kinds requirements name. */
  readonly accepts: ReadonlySet<string>;
  /** Kind constraints contributed by param decorators (e.g. @CurrentSession() → { kind: 'session', reason: 'SESSION_REQUIRED' }); checked at boot (B15). */
  readonly principalParams: readonly {
    readonly kind: string;
    readonly reason: string;
  }[];
  /**
   * The origin check (§7.10). 'cookie': better-auth's rule when the browser leg carries a cookie. 'form': better-auth's
   * form-CSRF rule, which also covers requests without a cookie (handlers with @ForwardAuthCookies(), such as login proxies).
   * 'off': @SkipOriginCheck() or originCheck.mode 'off'. [R2:SEC-r2-08]
   */
  readonly originCheck: "cookie" | "form" | "off";
  /** Content key of the sources this plan consults (ordered source indexes, interned per instance); part of the principal memo key (§7.2). [R2:NEST-r2-16] */
  readonly sourceSet: string;
  /** Set-Cookie of application direct calls is forwarded (§7.5). */
  readonly forwardDirectCalls: boolean;
  readonly site: string; // 'ProjectsController.remove'
  // v3's routeParams is gone: the named inputs of a handler (route params, GraphQL @Args names) come from the transports' claims
  // (ClaimOptions.inputs), so core reads no transport metadata for them (§4.2.1, BootAdviceContext.handlers). [R3:NEST-r3-10]
}

/** Opens the per-invocation scope used by the cookie bridge, refresh suppression, param decorators and the service. */
export declare class BetterAuthScopeInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<unknown>;
}

/**
 * Applies BetterAuthGuard + BetterAuthScopeInterceptor explicitly (every gateway, hybrid RPC controllers, globalGuard: false).
 * Handlers a transport claims as reachable only explicitly need both enhancers; B16 names this decorator (§5.4). [R2:NEST-r2-11]
 */
export declare function UseBetterAuth(): ClassDecorator & MethodDecorator;

export declare class BetterAuthService<A extends AuthLike = RegisteredAuth> {
  readonly name: string;
  /** The better-auth instance, unchanged and fully typed. */
  readonly instance: A;
  /**
   * instance.api: plain better-auth server calls (§7.5: forwarding is opt-in; refresh is suppressed in cookie-less scopes; inside a handler
   * scope, a call that carries the caller's session cookie on a browser leg — any method, whenever the plan's origin check is not 'off' —
   * whose origin no guard verified and for which no guard resolved the principal, as in a @Public() handler, throws
   * PUBLIC_HANDLER_USED_CALLER_SESSION). [R4:SEC-r4-02] [R5:SEC-r5-02]
   */
  readonly api: A["api"];
  /** Memoized await of instance.$context (resolves immediately after boot). */
  context(): Promise<Awaited<A["$context"]>>;
  /** Mount information after bootstrap; undefined when not mounted. */
  mount():
    | {
        readonly basePath: string;
        readonly bodyLimit: number;
        readonly platform: string;
      }
    | undefined;
  /**
   * Principal of the current handler scope: the reading the interceptor took for this invocation (§7.8). It never starts a
   * resolution (no better-auth I/O), so a public handler gets null (§7.6), and a handler whose guard did not run throws NO_AUTH_RESULT.
   * Outside any handler scope (guards, middleware, policies, background jobs) it throws a request-time configuration error
   * NO_AUTH_SCOPE: that is never mistaken for "anonymous". [R2:NEST-r2-07, SEC-r2-04]
   */
  getPrincipal(): Promise<AuthPrincipal | null>;
  /** The `session` field of the current principal when it carries one (session principals), else null. Same scope rules as getPrincipal(). */
  getSession(): Promise<SessionOf<A> | null>;
  /**
   * Synchronous reader for guards and interceptors. Never performs I/O, and never answers "don't know" (§7.8):
   *   public plans → { outcome: 'no-identity' };
   *   inherit plans → the reading of the nearest enclosing invocation (lineage), whatever its instance, else a test stamp, else NO_AUTH_RESULT;
   *   other plans → the result BetterAuthGuard produced for this invocation, else a test stamp, else it throws
   *   BetterAuthConfigurationError.atRequest('PRINCIPAL_READ_BEFORE_GUARD'): the calling guard runs before BetterAuthGuard (an
   *   APP_GUARD declared in AppModule runs before the one BetterAuthModule.forRoot() contributes) or BetterAuthGuard does not reach
   *   the handler. v3 returned undefined there, which also meant "public". [R2:NEST-r2-07] [R3:NEST-r3-02]
   * `instance` names the instance the reading belongs to: the plan's own, or for an inherit plan the enclosing reading's, whichever
   * instance this service serves. [R4:NEST-r4-03]
   */
  principalFor(context: ExecutionContext): PrincipalReading;
  /**
   * Web Headers for a direct auth.api call from a platform request: pseudo-headers dropped, client-IP header set. A handler that passes them
   * to a call acting on the caller's session uses @OptionalAuth() or a guarded access, never @Public() — on any route, safe methods included
   * (§7.5). [A] [R4:SEC-r4-02] [R5:SEC-r5-02]
   */
  headersFrom(platformRequest: unknown): Headers;
  /**
   * Inside a handler whose plan forwards direct calls (@ForwardAuthCookies() or cookies.forwardDirectCalls), also forward the
   * Set-Cookie of calls in fn that carry ANOTHER credential: deliberate account switching (impersonation, "sign in as"). Anywhere
   * else it throws BetterAuthConfigurationError.atRequest('FORWARDING_NOT_DECLARED'): forwarding is declared on the handler, where the
   * guard applies better-auth's form-CSRF rule to it (§7.5, §7.10). Replaces v3's forwardCookies(fn), which enabled forwarding in any
   * handler and so skipped that check on public login proxies. [R1:SEC-r1-03] [R3:SEC-r3-04]
   */
  forwardForeignCookies<T>(fn: () => Promise<T>): Promise<T>;
  /** Run fn in a cookie-less scope: every auth.api call inside has session refresh suppressed (§7.4). [C] */
  withoutCookies<T>(fn: () => Promise<T>): Promise<T>;
  /** Run fn outside the current scope: no forwarding, no suppression (plain better-auth). [C] */
  runOutsideScope<T>(fn: () => Promise<T>): Promise<T>;
}

/** What principalFor(ctx) reads: the guard's result, or 'no-identity' where the plan resolves none (public plans, and inherit plans under a public operation). `instance`: the instance whose guard run recorded it. [R3:NEST-r3-02] [R4:NEST-r4-03] */
export type PrincipalReading = (
  PrincipalResult | { readonly outcome: "no-identity" }
) & { readonly instance: string };

/**
 * The resolver port the guard uses. Default: ChainPrincipalResolver. Readers never call it: they read the results the guard
 * recorded (§7.8), so v3's peek() is gone. [B] [R3:NEST-r3-01]

export interface PrincipalResolver {
  resolve(
    call: TransportCall,
    request: ResolutionRequest,
  ): Promise<PrincipalResult>;
}
export interface ResolutionRequest {
  readonly auth: AuthHandle;
  readonly freshness: "default" | "authoritative";
  /** Kinds the route accepts; sources producing none of them are not consulted (§7.6). */
  readonly accepts: ReadonlySet<string>;
  /** The plan's content key of those sources (RoutePlan.sourceSet). */
  readonly sourceSet: string;
  /**
   * The re-classification read of a generic 401 a policy call met (§8.1): resolve again through the one source that produced the
   * principal (its id), which must declare sessionBacked, bypassing the principal memo. The evaluator shares one such read per logical
   * request (§7.2). [R2:BA-r2-03] [R3:BA-r3-01] [R4:SEC-r4-04]
   */
  readonly reclassify?: { readonly sourceId: string };
}

/** The port the evaluator calls a policy through. Default: policy.evaluate(params, context). Tests substitute it (overrideDecisions, §14.8). [R2:NEST-r2-15] */
export interface PolicyInvoker {

    policy: AuthorizationPolicy<P, any>,
    params: P,
